cmake_minimum_required(VERSION 3.0.2)
project(omnicopter_slam)
find_package(catkin REQUIRED COMPONENTS roscpp std_srvs nav_msgs sensor_msgs omni_firmware maneuver)
find_package(Threads REQUIRED)
catkin_package(CATKIN_DEPENDS roscpp std_srvs nav_msgs sensor_msgs omni_firmware maneuver)
add_executable(flight_demo src/flight_demo.cpp)
target_compile_features(flight_demo PRIVATE cxx_std_14)
target_include_directories(flight_demo PRIVATE ${catkin_INCLUDE_DIRS})
target_link_libraries(flight_demo ${catkin_LIBRARIES} Threads::Threads)
add_dependencies(flight_demo ${catkin_EXPORTED_TARGETS})
add_executable(localization_inputs src/localization_inputs.cpp)
target_compile_features(localization_inputs PRIVATE cxx_std_14)
target_include_directories(localization_inputs PRIVATE ${catkin_INCLUDE_DIRS})
target_link_libraries(localization_inputs ${catkin_LIBRARIES})
add_dependencies(localization_inputs ${catkin_EXPORTED_TARGETS})

install(TARGETS flight_demo localization_inputs RUNTIME DESTINATION ${CATKIN_PACKAGE_BIN_DESTINATION})
catkin_install_python(PROGRAMS test/pose_dropout.py DESTINATION ${CATKIN_PACKAGE_BIN_DESTINATION})
install(DIRECTORY launch config DESTINATION ${CATKIN_PACKAGE_SHARE_DESTINATION})
